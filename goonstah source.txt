sector.element('Button', 'Load Tool', nil, function()
    local main = getgenv().Goonstah.Main
    local aim = getgenv().Goonstah.Aim

    if main.LockToolLoaded then
        return
    end

    main.LockToolLoaded = true

    local Players = game:GetService("Players")
    local player = Players.LocalPlayer
    local Camera = workspace.CurrentCamera

    -- function to get closest target to screen center
    local function GetClosestToCenter()
        local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        local closest, minDist = nil, math.huge

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= player and plr.Character then
                local root = plr.Character:FindFirstChild("HumanoidRootPart")
                if root then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(root.Position)
                    if onScreen then
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                        if dist < minDist then
                            closest = plr
                            minDist = dist
                        end
                    end
                end
            end
        end
        return closest
    end

    -- clean toggle system
    local function ToggleTargetLockSafe()
        if not main.Enabled or not aim.TargetLockEnabled then
            aim.CurrentTarget = nil
            return
        end

        -- toggle on/off
        if aim.CurrentTarget then
            aim.CurrentTarget = nil
        else
            aim.CurrentTarget = GetClosestToCenter()
        end
    end

    local LockTool = Instance.new("Tool")
    LockTool.RequiresHandle = false
    LockTool.Name = "Lock Tool"
    LockTool.Parent = player:WaitForChild("Backpack")

    LockTool.Activated:Connect(function()
        if not main.Enabled then
            aim.CurrentTarget = nil
            if aim.CameraLockEnabled then aim.CameraLockEnabled = false end
            return
        end

        if aim.CameraLockEnabled and ToggleCameraLock then
            ToggleCameraLock()
        end

        if aim.TargetLockEnabled then
            ToggleTargetLockSafe()
        else
            aim.CurrentTarget = nil
        end
    end)

    -- stop locks if toggles off
    task.spawn(function()
        while task.wait(0.2) do
            if not main.Enabled or not aim.TargetLockEnabled then
                aim.CurrentTarget = nil
            end
        end
    end)

    -- persist tool
    player.CharacterAdded:Connect(function()
        task.wait(1)
        if not LockTool.Parent or LockTool.Parent ~= player.Backpack then
            LockTool.Parent = player.Backpack
        end
    end)
end)

		local speed = tonumber(ts.Speed) or 20

		local fakeCF
		if targetRoot then
			if ts.Randomized then
				local speedFactor = math.clamp(speed / 20, 0.5, 3)
				rT = math.min(1, rT + (dt * rRate * speedFactor))
				if rT >= 1 then
					rLast = rNext
					rNext = nextRandom(distance, height)
					rT = 0
				end
				local off = rLast:Lerp(rNext, rT)
				fakeCF = CFrame.new(targetRoot.Position + Vector3.new(off.X, height + off.Y, off.Z))
			else
				spin = spin + (speed * 2)
				fakeCF = targetRoot.CFrame * CFrame.Angles(0, math.rad(spin), 0) * CFrame.new(0, height, distance)
			end
		else
			-- not locked: dummy sits on you
			fakeCF = myRoot.CFrame
		end

		-- store real CF before spoof
		Desync[1] = myRoot.CFrame

		if ts.Spoof then
			local dummy = ensureDummy()
			if dummy and dummy.PrimaryPart then
				pcall(function() dummy:SetPrimaryPartCFrame(fakeCF) end)
			end

			-- replicate fake CFrame (others see this)
			pcall(function() myRoot.CFrame = fakeCF end)

			-- revert instantly (you stay normal)
			RunService.RenderStepped:Wait()
			pcall(function() myRoot.CFrame = Desync[1] end)
		else
			pcall(function() myRoot.CFrame = fakeCF end)
			hideDummy()
		end
	end)

	-----------------------------------------------------
	-- // Safe __index hook (stops visual pop)
	-----------------------------------------------------
	local oldIndex
	oldIndex = hookmetamethod(game, "__index", newcclosure(function(self, key)
		if not checkcaller() and key == "CFrame" and Desync[1]
			and Goonstah.TargetStrafe
			and Goonstah.TargetStrafe.Enabled
			and Goonstah.TargetStrafe.Spoof
			and LocalPlayer.Character
			and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
			and self == LocalPlayer.Character.HumanoidRootPart
		then
			return Desync[1]
		end
		return oldIndex(self, key)
	end))
end

-- 🔍 Find closest player to center
local function GetClosestPlayer()
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local closest, distance = nil, math.huge

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local root = player.Character.HumanoidRootPart
            local pos, onScreen = Camera:WorldToViewportPoint(root.Position)
            if onScreen then
                local diff = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                if diff < distance then
                    closest = player
                    distance = diff
                end
            end
        end
    end
    return closest
end

-- 🎥 Smooth camera move
local function MoveCameraToTarget(targetPos)
    local settings = Goonstah.CameraLock
    local smoothness = tonumber(settings.Smoothness) or 0.25
    local easingStyle = Enum.EasingStyle[settings.EasingStyle or "Linear"]
    local easingDir = Enum.EasingDirection[settings.EasingDirection or "In"]

    if smoothness >= 1 then
        Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetPos)
    else
        local tween = TweenService:Create(Camera, TweenInfo.new(smoothness, easingStyle, easingDir), {
            CFrame = CFrame.new(Camera.CFrame.Position, targetPos)
        })
        tween:Play()
    end
end

-- ⚙️ True predictive math using velocity and ping
local function GetPredictedPosition(part)
    if not part or not part.Parent then return part and part.Position end

    local hrp = part.Parent:FindFirstChild("HumanoidRootPart")
    if not hrp then return part.Position end

    local velocity = hrp.Velocity
    local horiz = tonumber(Goonstah.Aim.HorizontalPrediction) or 0
    local vert = tonumber(Goonstah.Aim.VerticalPrediction) or 0
    local jump = tonumber(Goonstah.Aim.JumpOffset) or 0
    local pingAdd = 0

    if Goonstah.Aim.PingBased and typeof(LocalPlayer.GetNetworkPing) == "function" then
        local ping = LocalPlayer:GetNetworkPing()
        pingAdd = ping * 0.1
    end

    -- ✅ actual predictive aim
    local predicted = hrp.Position + (velocity * (horiz + pingAdd))
    predicted = predicted + Vector3.new(0, vert + jump, 0)
    return predicted
end

    end)

-----------------------------------------------------
-- 🧰 Helpers
-----------------------------------------------------
local function safeIsAlive(pl)
    if not pl or not pl.Character then return false end
    local hrp = pl.Character:FindFirstChild("HumanoidRootPart")
    local hum = pl.Character:FindFirstChildOfClass("Humanoid")
    return hrp and hum and hum.Health > 0
end

-- ✅ Unified Velocity-Based Prediction
local function PredictPosition(part)
    if not part or not Goonstah.Aim then return nil end
    local hrp = part.Parent:FindFirstChild("HumanoidRootPart")
    if not hrp then return part.Position end

    local velocity = hrp.Velocity
    local horiz = tonumber(Goonstah.Aim.HorizontalPrediction) or 0
    local vert = tonumber(Goonstah.Aim.VerticalPrediction) or 0
    local jump = tonumber(Goonstah.Aim.JumpOffset) or 0
    local pingAdd = 0

    if Goonstah.Aim.PingBased and typeof(LocalPlayer.GetNetworkPing) == "function" then
        local ping = LocalPlayer:GetNetworkPing()
        pingAdd = ping * 0.1
    end

    local predicted = hrp.Position + (velocity * (horiz + pingAdd))
    predicted = predicted + Vector3.new(0, vert + jump, 0)

    if Goonstah.Aim.AntiAimView then
        local yaw = math.rad((math.random() - 0.5) * 6)
        predicted = (CFrame.new(predicted) * CFrame.Angles(0, yaw, 0)).p
    end

    return predicted
end

-----------------------------------------------------
-- 🎯 Silent Target Finder
-----------------------------------------------------
local function getSilentTarget()
    if not Goonstah.Aim then return nil end
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local best, bestDist = nil, math.huge
    local fov = tonumber(Goonstah.Aim.FOVSize) or 200

    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= LocalPlayer and safeIsAlive(pl) then
            local part = pl.Character:FindFirstChild(Goonstah.Aim.AimPart or "Head") or pl.Character:FindFirstChild("Head")
            if part then
                local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local d = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if d <= fov and d < bestDist then
                        best, bestDist = pl, d
                    end
                end
            end
        end
    end

    return best
end

Goonstah.Aim.SetSilentAimHook = function(fn)
    if not Goonstah.Aim then return end
    Goonstah.Aim.SilentAimHook = type(fn) == "function" and fn or nil
end

-----------------------------------------------------
-- 🔧 Hooks (Silent Aim + Target Lock | Universal MainEvent Support)
-----------------------------------------------------
do
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local Mouse = LocalPlayer:GetMouse()
    local MainEvent = ReplicatedStorage:FindFirstChild("MainEvent")
    local mt = getrawmetatable(game)
    local oldNamecall = mt.__namecall
    setreadonly(mt, false)

    local validArgNames = {
        "UpdateMousePos","GetMousePos","MousePos","MOUSE","MousePosUpdate",
        "UpdateMousePosI","UpdateMousePosI2","UpdateAim","UpdateAimingAt",
        "D3RHooDMSOUEPoS233^+","MOUSEPOSUPDATEI12","DEAHOODMOUSEPOSx3^3"
    }

    local cachedPos, lastUpdate = nil, 0
    local updateInterval = 0.05
    local ToolConn

    -----------------------------------------------------
    -- 🔮 Velocity Prediction (shared)
    -----------------------------------------------------
    local function PredictFromPart(part)
        return part and PredictPosition(part) or nil
    end

    -----------------------------------------------------
    -- 🎯 Target Resolver
    -----------------------------------------------------
    local function ResolveTarget()
        if not Goonstah.Main.Enabled then
            cachedPos = nil
            return nil
        end

        if Goonstah.Aim.TargetLockEnabled and Goonstah.Aim.CurrentTarget and Goonstah.Aim.CurrentTarget.Character then
            local t = Goonstah.Aim.CurrentTarget
            if t.Character and t.Character:FindFirstChild(Goonstah.Aim.AimPart or "Head") then
                return t
            end
        end

        if not Goonstah.Aim.SilentMode then return nil end
        return getSilentTarget()
    end

    -----------------------------------------------------
    -- 🧩 Tool Hook (Anti-Aim View compatible)
    -----------------------------------------------------
    local function Bypass(character)
        character.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then
                if ToolConn then ToolConn:Disconnect() end
                ToolConn = child.Activated:Connect(function()
                    if not (MainEvent and Goonstah and Goonstah.Aim and Goonstah.Aim.AntiAimView) then return end

                    local sendPos = Mouse.Hit.Position
                    local target = ResolveTarget()
                    if target and target.Character then
                        local part = target.Character:FindFirstChild(Goonstah.Aim.AimPart or "Head")
                        if part then
                            sendPos = PredictFromPart(part)
                        end
                    end
                    MainEvent:FireServer("UpdateMousePos", sendPos)
                end)
            end
        end)
    end

    LocalPlayer.CharacterAdded:Connect(Bypass)
    if LocalPlayer.Character then Bypass(LocalPlayer.Character) end

    -----------------------------------------------------
    -- 🧩 MainEvent Hook
    -----------------------------------------------------
    mt.__namecall = newcclosure(function(self, ...)
        local args = {...}
        local method = getnamecallmethod()

        if checkcaller() or not Goonstah.Main.Enabled or not (Goonstah.Aim.UseNamecall or Goonstah.Aim.SilentMode or Goonstah.Aim.TargetLockEnabled or Goonstah.Aim.AntiAimView) then
            return oldNamecall(self, ...)
        end

        if method == "FireServer" then
            local remoteName = tostring(self):lower()
            if remoteName:match("mainevent") or remoteName:match("mainremoteevent") then
                local stringArgIndex
                for i, arg in ipairs(args) do
                    if typeof(arg) == "string" and table.find(validArgNames, arg) then
                        stringArgIndex = i
                        break
                    end
                end

                if stringArgIndex then
                    local now = tick()
                    if now - lastUpdate > updateInterval then
                        lastUpdate = now
                        local target = ResolveTarget()
                        if target and target.Character then
                            local part = target.Character:FindFirstChild(Goonstah.Aim.AimPart or "Head") or target.Character:FindFirstChild("Head")
                            cachedPos = part and PredictFromPart(part)
                        else
                            cachedPos = nil
                        end
                    end

                    if cachedPos then
                        args[stringArgIndex + 1] = cachedPos
                    end
                end
            end
        end
        return oldNamecall(self, unpack(args))
    end)

    setreadonly(mt, true)
end


-- 🌀 Main loop
do
    local lastTarget = nil
    RunService.RenderStepped:Connect(function()
        if not (Goonstah.Main and Goonstah.Main.Enabled) or not (Goonstah.Aim and Goonstah.Aim.SilentMode) then
            if Goonstah.Aim then
                Goonstah.Aim._UpdateFOV(Goonstah.Aim.FOVSize, false)
            end
            lastTarget = nil
            return
        end

        if Goonstah.Aim then
            Goonstah.Aim._UpdateFOV(Goonstah.Aim.FOVSize, Goonstah.Aim.FOVShow)
        end

        local target = getSilentTarget()
        if target and target ~= lastTarget then
            lastTarget = target
            if type(Goonstah.Notify) == "function" then
                pcall(Goonstah.Notify, "Targeting (Silent): "..tostring(target.DisplayName or target.Name))
            else
                print("[Goonstah] Silent locked on:", target.Name)
            end
        end
    end)
end

-- // Main TriggerBot loop
RunService.RenderStepped:Connect(function()
    local TB = getgenv().Goonstah.TriggerBot or {}
    if not TB.Enabled then
        Circle.Visible = false
        return
    end

    local radius = tonumber(TB.FOV) or 100
    local delay = tonumber(TB.Delay) or 0.15

    -- Circle stays invisible and small, no resizing
    Circle.Visible = true
    Circle.Size = UDim2.new(0, 10, 0, 10) 

    -- Center logic for calculations
    local center = Camera.ViewportSize * 0.5

    -- Rate limit
    local now = tick()
    if now - lastTriggerTime < delay then return end

    -- Find closest visible part in FOV
    local closestPart, minDistance = nil, radius
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            for _, partName in ipairs({"Head", "HumanoidRootPart", "LowerTorso"}) do
                local part = player.Character:FindFirstChild(partName)
                if part then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                    if onScreen then
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                        if dist < minDistance then
                            -- Wall check
                            rayParams.FilterDescendantsInstances = {LocalPlayer.Character}
                            local origin = Camera.CFrame.Position
                            local direction = part.Position - origin
                            local result = workspace:Raycast(origin, direction, rayParams)
                            if result and not result.Instance:IsDescendantOf(player.Character) then
                                continue
                            end
                            minDistance = dist
                            closestPart = part
                        end
                    end
                end
            end
        end
    end

    -- Activate tool if a valid target is found
    if closestPart then
        local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
        if tool and not tool.Name:lower():find("katana") and not tool.Name:lower():find("knife") then
            lastTriggerTime = tick()
            task.spawn(function()
                tool:Activate()
                task.wait(delay / 2)
                tool:Deactivate()
            end)
        end
    end
end)
