local KeySystem = {

        ["marcella1"] = {tier = "admin", expires = nil},
        ["marcellaxd"] = {tier = "premium", expires = nil}

    authenticated = false,
    maxAttempts = 3,
    currentAttempts = 0,
    userTier = "none",
    sessionToken = nil,
    displayKey = "marcella"
}

local function generateSessionToken()
    local chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789"
    local token = ""
    for i = 1, 32 do
        local rand = math.random(1, #chars)
        token = token .. chars:sub(rand, rand)
    end
    return token
end

local function verifyKey(inputKey)
    KeySystem.currentAttempts = KeySystem.currentAttempts + 1

    if KeySystem.validKeys[inputKey] then
        KeySystem.authenticated = true
        KeySystem.userTier = KeySystem.validKeys[inputKey].tier
        KeySystem.sessionToken = generateSessionToken()
        return true
    end

    if KeySystem.currentAttempts >= KeySystem.maxAttempts then
        wait(2)
        game.Players.LocalPlayer:Kick("\n\nXFC Security System\nAccess Denied\n\nToo many failed attempts")
        return false
    end

    return false
end

local function createParticleEffect(parent)
    local particles = {}

    for i = 1, 15 do
        local particle = Instance.new("Frame")
        particle.Name = "Particle_" .. i
        particle.Parent = parent

        local grayValue = math.random(120, 200)
        particle.BackgroundColor3 = Color3.fromRGB(grayValue, grayValue, grayValue)
        particle.BorderSizePixel = 0
        particle.Size = UDim2.new(0, math.random(2, 3), 0, math.random(2, 3))
        particle.Position = UDim2.new(
            math.random(0, 100) / 100,
            0,
            math.random(0, 100) / 100,
            0
        )
        particle.BackgroundTransparency = math.random(50, 85) / 100
        particle.ZIndex = 10

        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(1, 0)
        corner.Parent = particle

        table.insert(particles, particle)

        spawn(function()
            while particle.Parent do
                local randomX = math.random(-40, 40)
                local randomY = math.random(-40, 40)

                particle:TweenPosition(
                    UDim2.new(
                        particle.Position.X.Scale + randomX/1000,
                        0,
                        particle.Position.Y.Scale + randomY/1000,
                        0
                    ),
                    Enum.EasingDirection.InOut,
                    Enum.EasingStyle.Sine,
                    math.random(25, 35) / 10,
                    false
                )

                wait(math.random(25, 35) / 10)
            end
        end)
    end

    return particles
end

    local function isPlayerDead(player)
        if not player.Character then return true end
        local humanoid = player.Character:FindFirstChild("Humanoid")
        if not humanoid then return true end
        return humanoid.Health <= 0
    end

    local function isOnSameTeam(player)
        if not LocalPlayer.Team or not player.Team then return false end
        return LocalPlayer.Team == player.Team
    end

    local function getPlayerFromPart(part)
        local character = part:FindFirstAncestorOfClass("Model")
        if character then
            local player = Players:GetPlayerFromCharacter(character)
            return player
        end
        return nil
    end

    local lastShot = 0
    local shooting = false

    RunService.RenderStepped:Connect(function()
        if not config.enabled or shooting then return end

        local target = Mouse.Target
        if not target then return end

        local player = getPlayerFromPart(target)
        if not player or player == LocalPlayer then return end

        -- Team Check
        if config.teamCheck and isOnSameTeam(player) then return end

        -- Dead Check
        if config.ignoreDeadPlayers and isPlayerDead(player) then return end

        -- Check if hovering over player character
        if player.Character and target:IsDescendantOf(player.Character) then
            local currentTime = tick()
            if currentTime - lastShot >= config.delay / 1000 then
                shooting = true
                lastShot = currentTime

                -- Click
                mouse1press()
                wait(0.01)
                mouse1release()

                shooting = false
            end
        end
    end)
